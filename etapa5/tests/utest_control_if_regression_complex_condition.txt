// Teste de Regressão: IF com condições complexas que geram código
// Teste para garantir que condições complexas funcionam corretamente
// Deve gerar código completo mesmo com expressões aninhadas

main -> inteiro := [
    var x := inteiro
    var y := inteiro
    var z := inteiro
    x := 5
    y := 10
    z := 15
    // IF com expressão aritmética complexa na condição
    se (x + y * 2 > z) [
        x := 100
    ]
    // IF com múltiplas operações na condição
    se ((x + y) * 2 < z * 3) [
        x := 200
    ]
    // IF com operadores lógicos na condição
    se (x > 0 & y > 0 & z > 0) [
        x := 300
    ]
];
