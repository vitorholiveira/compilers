    var x := inteiro
    x := 5
    // IF-ELSE com then vazio
    se (x < 10) [

    ]
    // IF-ELSE com else vazio
    se (x > 0) [
        x := 20
    ] senao [
    ]
    // IF-ELSE com ambos vazios
    se (x == 5) [
    ] senao [
    ]
