// Teste: IF com todos os operadores de comparação
// Teste para verificar que todos os operadores de comparação funcionam em IF
// Deve gerar cmp_LT, cmp_LE, cmp_EQ, cmp_GE, cmp_GT, cmp_NE

main -> inteiro := [
    var x := inteiro
    var y := inteiro
    x := 5
    y := 10
    se (x < y) [
        x := 1
    ]
    se (x <= y) [
        x := 2
    ]
    se (x == y) [
        x := 3
    ]
    se (x >= y) [
        x := 4
    ]
    se (x > y) [
        x := 5
    ]
    se (x != y) [
        x := 6
    ]
];
